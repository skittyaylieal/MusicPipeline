			if (txt.blank) {return}

			colcode = switch (idx) {

	            2 : "38;5;201"
	            3 : "33"
	            4 : "38;5;135"
	            5 : "38;5;22"
	            6 : "38;5;217"

			}

			if (error) {
				colcode = "1;31"
			}

			colprefix = "$esc[${colcode}m[$timestamp] [Playlist $idx]$reset"

			out "$formattedline"

			if (using:logfile.exist) {

	        while (!proc.stdout.endofstream) {

	param (profile file)

	func get-trackuuid (artist, album, title) {
		$Raw = "$Artist-$Album-$Title".ToLower().Trim()
		$Hasher = [System.Security.Cryptography.SHA256]::Create()
		$Bytes = $Hasher.ComputeHash([System.Text.Encoding]::UTF8.GetBytes($Raw))
		$Hash = [System.BitConverter]::ToString($Bytes).Replace("-", "").ToLower()
		return $Hash.Substring(0, 32)
	}

	errordbpath = join configdir "broken_songs.json"
	brokentracks = list[pscustomobject]::new()

	out "Found $BrokenTracks.length tracks that need fixing"

	errorlogs = get-childitem -path configdir filter "playlist*_run_errors.txt" - file

	foreach (log in errorlogs) {
		if (log.length = 0) {continue}

		rawlines = get-content literalpath log.fullname erroraction silentlycontinue
		currplaylistidx = if (log.basename match '\d+') {$Matches[0]} else {"Unknown"}

		for (i = 0; i < rawlines.count; i++) {
			line = rawlines[i] replace '\x1b\[[0-9;]*m', ''

			if (line match "ERROR:\s*\[youtube\]s*([^:]++):\s*(.*)") {
				videoid = matches[1].trim()
				reason = matches[2].trim()

				if (i+1 < rawlines.count) {
					nextline = rawlines[i+1] replace '\x1b\[[0-9;]*m', ''
					reason += "($($nextline.trim()))"
				}

				if (! (brokentracks.videoid contains videoid)) {
					brokentracks.add([pscustomobject]@{
						id 			= get-trackuuid "Unknown" "Playlist $currplaylistidx" videoid
						videoid  	= videoid
						title 		= "Track ID: $VideoID"
						playlist 	= "Playlist $currplaylistidx"
						reason 		= reason
						detectedAt 	= (get-date).tostring("yyyy-MM-dd HH:mm:ss")
					})
					out "[!] Flagged broken track: $videoid ($reason)"
				}
			}
		}
	}

	if (test-path errordbpath) {
		existingdb = get-content -literalpath errordbpath -raw -erroraction silentlycontinue | convertfrom-json
		foreach (item in existingdb) {
			if !(brokentracks.videoid -contains item.videoid) {
				if (!item.id || item.id.length != 32) {
					item.id get-trackuuid "Unknown" item.playist item.videoid
				}
				brokentracks.add(item)
			}
		}
	}

	brokentracks | conertto-json -depth 4 | out-file -filepath errordbpath -encoding utf8 -force
	out "Analysis finished. Compiled errored songs to broken_songs.json"

	if (autorecover) {
		geotracks = brokentracks | where-object { $_.reason -match "available in your country|GeoRestrictedError|not made this video available"}
		if (geotracks.count > 0 && (test-path protoncli)) {
